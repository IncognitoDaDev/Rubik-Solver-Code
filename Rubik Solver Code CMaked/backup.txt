// helper function:
// finds a cosine of angle between vectors
// from pt0->pt1 and from pt0->pt2
static double angle(Point pt1, Point pt2, Point pt0)
{
	double dx1 = pt1.x - pt0.x;
	double dy1 = pt1.y - pt0.y;
	double dx2 = pt2.x - pt0.x;
	double dy2 = pt2.y - pt0.y;
	return (dx1 * dx2 + dy1 * dy2) / sqrt((dx1 * dx1 + dy1 * dy1) * (dx2 * dx2 + dy2 * dy2) + 1e-10);
}

Point MiddlePoint(vector<Point> sq) // Calculates the center point of the square
{
    Point center;
    center.x = (sq[0].x + sq[3].x) / 2;
    center.y = (sq[0].y + sq[3].y) / 2;
    return center;
}

bool isOverlapping(vector<Point> sq1, vector<Point> sq2) //Checks if the two squares is either above the other
{
    int l = (sq1[2].x - sq1[0].x) + tolerance;
    int xDist = abs(MiddlePoint(sq1).x - MiddlePoint(sq2).x),
        yDist = abs(MiddlePoint(sq1).y - MiddlePoint(sq2).y);

    return xDist <= l && yDist <= l;
}

// calculating and using the center point of the frame as point of reference
Point sqPosition(Mat frame, vector<Point> square)
{
    int centerX = frame.size().width / 2,
        centerY = frame.size().height / 2;

    // Every square is by default a centerpiece until proven otherwise...
    Point pos; pos.x = 1; pos.y = 1;

    int l = abs(square[0].x - square[1].x) + tolDist;

    if (centerX - l >= MiddlePoint(square).x) pos.x--; // Left Col
    else if (centerX + l <= MiddlePoint(square).x) pos.x++; // Right Col

    if (centerY - l >= MiddlePoint(square).y) pos.y--; // Down Row
    else if (centerY + l <= MiddlePoint(square).y) pos.y++; // Up Row

    return pos;
}

Mat ColorFilter(Mat image, int color)
{

    }

    case Blue:
    {
        cvtColor(image, hsv, COLOR_BGR2HSV);
        lowRange = Scalar(100, 90, 80);
        highRange = Scalar(120, 255, 255);
        break;
    }

    case Yellow:
    {
        cvtColor(image, hsv, COLOR_BGR2HSV);
        lowRange = Scalar(20, 160, 160);
        highRange = Scalar(30, 255, 255);
        break;
    }

    case Red: // Needs more refining...?
    {
        cvtColor(image, hsv, COLOR_RGB2HSV);
        lowRange = Scalar(118, 130, 70);
        highRange = Scalar(140, 255, 205);
        break;
    }

    case Green:
    {
        cvtColor(image, hsv, COLOR_BGR2HSV);
        lowRange = Scalar(50, 120, 70);
        highRange = Scalar(80, 255, 255);
        break;
    }

    case Orange:
    {
        cvtColor(image, hsv, COLOR_BGR2HSV);
        lowRange = Scalar(5, 205, 65);
        highRange = Scalar(20, 255, 255);
        break;
    }
    }
    inRange(hsv, lowRange, highRange, end);

    cvtColor(end, end, COLOR_GRAY2BGR);
    bitwise_and(image, end, end);

    return end;
}

Mat CombineColorFilterResults(Mat frame, vector<int> colorToFilter)
{
    Mat end(frame.size().height, frame.size().width, CV_8UC3, Scalar(0, 0, 0));
    for (int i : colorToFilter)
    {
        Mat sq2 = ColorFilter(frame, i), sq_gray, mask, bg, fg;
        cvtColor(sq2, sq_gray, COLOR_BGR2GRAY);
        threshold(sq_gray, mask, 120, 255, THRESH_BINARY);

        bitwise_or(end, end, mask);

        Mat maskinv; bitwise_not(sq_gray, maskinv);
        bitwise_and(sq2, sq2, fg, maskinv);

        add(end, fg, end);
    }

    return end;
}

static void findSquares(const Mat& image, vector<vector<Point>>& squares)
{
    squares.clear();

    Mat timg(image);
    //medianBlur(image, timg, 3);
    Mat gray0(timg.size(), CV_8U), gray;

    vector<vector<Point> > contours;

    // find squares in every color plane of the image
    for (int c = 0; c < 3; c++)
    {
        int ch[] = { c, 0 };
        mixChannels(&timg, 1, &gray0, 1, ch, 1);
        for (int l = 0; l < N; l++)
        {
            // hack: use Canny instead of zero threshold level.
            // Canny helps to catch squares with gradient shading
            if (l == 0)
            {
                // apply Canny. Take the upper threshold from slider
                // and set the lower to 0 (which forces edges merging)
                Canny(gray0, gray, 5, thresh, 5);
                // dilate canny output to remove potential
                // holes between edge segments
                dilate(gray, gray, Mat(), Point(-1, -1));
            }
            else
            {
                gray = gray0 >= (l + 1) * 255 / N;
            }

            // find contours and store them all as a list
            findContours(gray, contours, RETR_LIST, CHAIN_APPROX_SIMPLE);

            vector<Point> approx;

            // test each contour
            for (size_t i = 0; i < contours.size(); i++)
            {
                approxPolyDP(Mat(contours[i]), approx, arcLength(Mat(contours[i]), true) * 0.02, true);

                if (approx.size() == 4 &&
                    fabs(contourArea(Mat(approx))) > minSize &&
                    fabs(contourArea(Mat(approx))) < maxSize &&
                    isContourConvex(Mat(approx)))
                {
                    double maxCosine = 0;

                    for (int j = 2; j < 5; j++)
                    {
                        // find the maximum cosine of the angle between joint edges
                        double cosine = fabs(angle(approx[j % 4], approx[j - 2], approx[j - 1]));
                        maxCosine = MAX(maxCosine, cosine);
                    }

                    // if cosines of all angles are small
                    // (all angles are ~90 degree) then write quandrange
                    // vertices to resultant sequence
                    if (maxCosine < 0.3)
                        squares.push_back(approx);
                }
            }
        }
    }
}

static void findSquaresAlt(const Mat image, vector<vector<Point>>& squares, vector<int> &colorType)
{
    int N2 = 2;
    squares.clear();

    vector<vector<Point> > contours;

    // find squares in every rubiks cube's color possible;
    for (int c = 1; c < 6; c++)
    {
        for (int l = 0; l < N2; l++)
        {
            Mat end = ColorFilter(image, c);
            cvtColor(end, end, COLOR_BGR2GRAY);
            //medianBlur(end, end, 5);
            dilate(end, end, Mat(), Point(-1, -1));

            findContours(end, contours, RETR_LIST, CHAIN_APPROX_SIMPLE);

            vector<Point> approx;

            // test each contour
            for (size_t i = 0; i < contours.size(); i++)
            {
                approxPolyDP(Mat(contours[i]), approx, arcLength(Mat(contours[i]), true) * 0.02, true);

                if (approx.size() == 4 &&
                    fabs(contourArea(Mat(approx))) > minSize &&
                    fabs(contourArea(Mat(approx))) < maxSize &&
                    isContourConvex(Mat(approx)))
                {
                    double maxCosine = 0;

                    for (int j = 2; j < 5; j++)
                    {
                        // find the maximum cosine of the angle between joint edges
                        double cosine = fabs(angle(approx[j % 4], approx[j - 2], approx[j - 1]));
                        maxCosine = MAX(maxCosine, cosine);
                    }

                    // if cosines of all angles are small
                    // (all angles are ~90 degree) then write quandrange
                    // vertices to resultant sequence
                    if (maxCosine < 0.3)
                    {
                        squares.push_back(approx);
                        colorType.push_back(c);
                    }
                }
            }
        }
    }
}

static void filterExtraContours(vector<vector<Point>> &sq, vector<int> &ct)
{
    if (sq.size() < 2) return;

    vector<int> validsq; 

    for (int i = 0; i < sq.size(); i++) validsq.push_back(1);
    
    cout << "Filtering out the colliding squares... ";

    for (int i = 0; i < sq.size() - 1; i++)
    {
        if (validsq[i] == 0) continue;

        for (int j = i + 1; j < sq.size(); j++)
            if (isOverlapping(sq[i], sq[j])) validsq[i] = 0;
    }

    vector<vector<Point>> nsq; 
    vector<int> nct;
    for (int i = 0; i < validsq.size(); i++)
    {
        if (validsq[i] == 1)
        {
            nsq.push_back(sq[i]);
            nct.push_back(ct[i]);
        }
    }
    
    cout << nsq.size() << " / " << sq.size() << endl;

    sq.clear(); sq = nsq;
    ct.clear(); ct = nct;
}

static void drawSquares(Mat& image, const vector<vector<Point>>& squares)
{
    for (size_t i = 0; i < squares.size(); i++)
    {
        const Point* p = &squares[i][0];

        int n = (int)squares[i].size();
        //dont detect the border
        if (p->x > 3 && p->y > 3)
            polylines(image, &p, &n, 1, true, Scalar((i * 27 % 255), 255, (i * 47 % 255)), 2, LINE_AA);
    }
}

void SortListSQ(vector<vector<Point>> &sq)
{
    if (sq.size() < 2) return;
    int l = sq[0][1].x - sq[0][0].x + tolerance;
    int ok = 0;
    while (ok == 0) // Sort by height
    {
        ok = 1;
        for (int i = 0; i < sq.size() - 1; i++)
            if (MiddlePoint(sq[i]).y < MiddlePoint(sq[i+1]).y)
            {
                vector<Point> temp = sq[i+1];
                sq[i+1] = sq[i];
                sq[i] = temp;
                ok = 0;
            }
    }

    ok = 0;
    while (ok == 0)
    {
        ok = 1;
        for (int i = 0; i < sq.size() - 1; i++)
            if (abs(MiddlePoint(sq[i]).y - MiddlePoint(sq[i + 1]).y) <= l)
            {
                if (MiddlePoint(sq[i]).x > MiddlePoint(sq[i + 1]).x)
                {
                    vector<Point> temp = sq[i + 1];
                    sq[i + 1] = sq[i];
                    sq[i] = temp;
                    ok = 0;
                }
            }
    }
}

void translateFaceColor(Mat &frame, vector<vector<Point>> square, vector<int> colorType, int (& ma)[3][3])
{
    // White being the default color for all squares
    for (int i = 0; i < 3; i++)
        for (int j = 0; j < 3; j++) ma[i][j] = 0;

    if (square.size() < 1) return;

    for (int i = 0; i < square.size(); i++)
    {
        Point pos = sqPosition(frame, square[i]);
        ma[pos.y][pos.x] = colorType[i];
    }

    int debugLines = 1;

    if (debugLines)
    {
        int centerY = frame.size().height / 2,
            centerX = frame.size().width / 2,
            l = abs(square[0][0].x - square[0][1].x) + tolDist;

        //Center lines
        line(frame, Point(frame.size().width, centerY), Point(0, centerY), Scalar(0, 0, 0), 1, 0);
        line(frame, Point(centerX, frame.size().height), Point(centerX, 0), Scalar(0, 0, 0), 1, 0);

        //Offset Lines
        line(frame, Point(frame.size().width, centerY - l), Point(0, centerY - l), Scalar(0, 0, 255), 1, 0);
        line(frame, Point(frame.size().width, centerY + l), Point(0, centerY + l), Scalar(0, 0, 255), 1, 0);
        line(frame, Point(centerX - l, frame.size().height), Point(centerX - l, 0), Scalar(0, 0, 255), 1, 0);
        line(frame, Point(centerX + l, frame.size().height), Point(centerX + l, 0), Scalar(0, 0, 255), 1, 0);
    }
}

void faceSwitch(int &ogfront, int &ogup, int front, int up, int(&rc)[6][3][3])
{
    int nrc[6][3][3];
    int left, right, bottom, back;
    identifyFaces(front, up, left, right, bottom, back);

    int ogleft, ogright, ogbottom, ogback;
    identifyFaces(ogfront, ogup, ogleft, ogright, ogbottom, ogback);

    int color[6] = { front, up, left, right, bottom, back },
        ogcolor[6] = { ogfront, ogup, ogleft, ogright, ogbottom, ogback };

    for (int l = 0; l < 6; l++)
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
            {
                nrc[color[l]][i][j] = rc[ogcolor[l]][i][j];
            }

    ogfront = front;
    ogup = up;
    copyRC(rc, nrc);
    

}

void faceRotate(int front, int up, int(&rc)[6][3][3], int anticlockwise, char turnType)
{
    int left, right, bottom, back;
    identifyFaces(front, up, left, right, bottom, back);

    int temp[12];
    switch (turnType)
    {
    case 'F': // FRONT TURNS
        // code for rotating a face by 90
        // first we move the data on front face
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[front][i][j];
                else a[2 - j][i] = rc[front][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[front][i][j] = a[i][j];

        if (anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][2 - i][2];
                    break;
                case 1: // up side
                    temp[i] = rc[up][2][i % 3];
                    rc[up][2][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][i % 3][0];
                    rc[right][i % 3][0] = temp[i - 3];
                    break;
                case 3: // down side
                    temp[i] = rc[bottom][0][2 - i % 3];
                    rc[bottom][0][2 - i % 3] = temp[i - 3];
                    rc[left][2 - i % 3][2] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][i][2];
                    break;
                case 1: // bottom side
                    temp[i] = rc[bottom][0][i % 3];
                    rc[bottom][0][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][2 - i % 3][0];
                    rc[right][2 - i % 3][0] = temp[i - 3];
                    break;
                case 3: // up side
                    temp[i] = rc[up][2][2 - i % 3];
                    rc[up][2][2 - i % 3] = temp[i - 3];
                    rc[left][i % 3][2] = temp[i];
                    break;
                }
            }
    } break;
    case 'B':
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[back][i][j];
                else a[2 - j][i] = rc[back][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[back][i][j] = a[i][j];

        if (anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][2 - i][0];
                    break;
                case 1: // up side
                    temp[i] = rc[up][0][i % 3];
                    rc[up][0][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][i % 3][2];
                    rc[right][i % 3][2] = temp[i - 3];
                    break;
                case 3: // down side
                    temp[i] = rc[bottom][2][2 - i % 3];
                    rc[bottom][2][2 - i % 3] = temp[i - 3];
                    rc[left][2 - i % 3][0] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][i][0];
                    break;
                case 1: // bottom side
                    temp[i] = rc[bottom][2][i % 3];
                    rc[bottom][2][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][2 - i % 3][2];
                    rc[right][2 - i % 3][2] = temp[i - 3];
                    break;
                case 3: // up side
                    temp[i] = rc[up][0][2 - i % 3];
                    rc[up][0][2 - i % 3] = temp[i - 3];
                    rc[left][i % 3][0] = temp[i];
                    break;
                }
            }
    } break;
    case 'U': // UP TURNS
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[up][i][j];
                else a[2 - j][i] = rc[up][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[up][i][j] = a[i][j];

        if (anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][0][i];
                    break;
                case 1: // front side
                    temp[i] = rc[front][0][i % 3];
                    rc[front][0][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][0][i % 3];
                    rc[right][0][i % 3] = temp[i - 3];
                    break;
                case 3: // back side
                    temp[i] = rc[back][0][2 - i % 3];
                    rc[back][0][2 - i % 3] = temp[i - 3];
                    rc[left][0][i % 3] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][0][2 - i];
                    break;
                case 1: // back side
                    temp[i] = rc[back][0][i % 3];
                    rc[back][0][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][0][2 - i % 3];
                    rc[right][0][2 - i % 3] = temp[i - 3];
                    break;
                case 3: // front side
                    temp[i] = rc[front][0][2 - i % 3];
                    rc[front][0][2 - i % 3] = temp[i - 3];
                    rc[left][0][i % 3] = temp[i];
                    break;
                }
            }
    } break;
    case 'D': // BOTTOM TURNS
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[bottom][i][j];
                else a[2 - j][i] = rc[bottom][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[bottom][i][j] = a[i][j];

        if (!anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][2][i];
                    break;
                case 1: // front side
                    temp[i] = rc[front][2][i % 3];
                    rc[front][2][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][2][i % 3];
                    rc[right][2][i % 3] = temp[i - 3];
                    break;
                case 3: // back side
                    temp[i] = rc[back][2][2 - i % 3];
                    rc[back][2][2 - i % 3] = temp[i - 3];
                    rc[left][2][i % 3] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // left side
                    temp[i] = rc[left][2][2 - i];
                    break;
                case 1: // back side
                    temp[i] = rc[back][2][i % 3];
                    rc[back][2][i % 3] = temp[i - 3];
                    break;
                case 2: // right side
                    temp[i] = rc[right][2][2 - i % 3];
                    rc[right][2][2 - i % 3] = temp[i - 3];
                    break;
                case 3: // front side
                    temp[i] = rc[front][2][2 - i % 3];
                    rc[front][2][2 - i % 3] = temp[i - 3];
                    rc[left][2][i % 3] = temp[i];
                    break;
                }
            }
    } break;
    case 'R':
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[right][i][j];
                else a[2 - j][i] = rc[right][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[right][i][j] = a[i][j];

        if (anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // front side
                    temp[i] = rc[front][2 - i][2];
                    break;
                case 1: // up side
                    temp[i] = rc[up][2 - i % 3][2];
                    rc[up][2 - i % 3][2] = temp[i - 3];
                    break;
                case 2: // back side
                    temp[i] = rc[back][i % 3][2];
                    rc[back][i % 3][2] = temp[i - 3];
                    break;
                case 3: // bottom side
                    temp[i] = rc[bottom][i % 3][2];
                    rc[bottom][i % 3][2] = temp[i - 3];
                    rc[front][2 - i % 3][2] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // front side
                    temp[i] = rc[front][i][2];
                    break;
                case 1: // down side
                    temp[i] = rc[bottom][i % 3][2];
                    rc[bottom][i % 3][2] = temp[i - 3];
                    break;
                case 2: // back side
                    temp[i] = rc[back][2 - i % 3][2];
                    rc[back][2 - i % 3][2] = temp[i - 3];
                    break;
                case 3: // up side
                    temp[i] = rc[up][2 - i % 3][2];
                    rc[up][2 - i % 3][2] = temp[i - 3];
                    rc[front][i % 3][2] = temp[i];
                    break;
                }
            }
    } break;
    case 'L':
    {
        int a[3][3];
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
                if (anticlockwise) a[j][2 - i] = rc[left][i][j];
                else a[2 - j][i] = rc[left][i][j];
        }

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) rc[left][i][j] = a[i][j];

        if (!anticlockwise)
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // front side
                    temp[i] = rc[front][2 - i][0];
                    break;
                case 1: // up side
                    temp[i] = rc[up][2 - i % 3][0];
                    rc[up][2 - i % 3][0] = temp[i - 3];
                    break;
                case 2: // back side
                    temp[i] = rc[back][i % 3][0];
                    rc[back][i % 3][0] = temp[i - 3];
                    break;
                case 3: // up side
                    temp[i] = rc[bottom][i % 3][0];
                    rc[bottom][i % 3][0] = temp[i - 3];
                    rc[front][2 - i % 3][0] = temp[i];
                    break;
                }
            }
        else
            for (int i = 0; i < 12; i++)
            {
                switch (i / 3)
                {
                case 0: // front side
                    temp[i] = rc[front][i][0];
                    break;
                case 1: // down side
                    temp[i] = rc[bottom][i % 3][0];
                    rc[bottom][i % 3][0] = temp[i - 3];
                    break;
                case 2: // back side
                    temp[i] = rc[back][2 - i % 3][0];
                    rc[back][2 - i % 3][0] = temp[i - 3];
                    break;
                case 3: // up side
                    temp[i] = rc[up][2 - i % 3][0];
                    rc[up][2 - i % 3][0] = temp[i - 3];
                    rc[front][i % 3][0] = temp[i];
                    break;
                }
            }
    } break;
    }
}

void actionRubik(int front, int up, int(&rc)[6][3][3], char action)
{
    int left, right, bottom, back;
    identifyFaces(front, up, left, right, bottom, back);

    switch (action)
    {
    default:
        cout << action << " is not a thing!" << endl;
        break;
    
        // normal moveset
    case 'U':
        faceRotate(front, up, rc, 0, 'U');
        break;
    case 'D':
        faceRotate(front, up, rc, 0, 'D');
        break;
    case 'L':
        faceRotate(front, up, rc, 0, 'L');
        break;
    case 'R':
        faceRotate(front, up, rc, 0, 'R');
        break;
    case 'F':
        faceRotate(front, up, rc, 0, 'F');
        break;
    case 'B':
        faceRotate(front, up, rc, 0, 'B');
        break;
        
        // inversed (normal) moveset (formally known as the ones with prime)
    case 'u':
        faceRotate(front, up, rc, 1, 'U');
        break;
    case 'd':
        faceRotate(front, up, rc, 1, 'D');
        break;
    case 'l':
        faceRotate(front, up, rc, 1, 'L');
        break;
    case 'r':
        faceRotate(front, up, rc, 1, 'R');
        break;
    case 'f':
        faceRotate(front, up, rc, 1, 'F');
        break;
    case 'b':
        faceRotate(front, up, rc, 1, 'B');
        break;

        // cube rotations (not to be confused with substited letters for prime moves)
    case 'x': // rotate the front face to up face
        faceSwitch(front, up, up, back, rc);
        break;
    case 'y': // rotate the right face to front face
        faceSwitch(front, up, left, up, rc);
        break;
    case 'z': // rotate the up face to right face
        faceSwitch(front, up, front, right, rc);
        break;
    }
}

void ReadRubik(int front, int up, int rc[6][3][3])
{
    int left, right, bottom, back;
    identifyFaces(front, up, left, right, bottom, back);

    // up face
    for (int y = 0; y < 3; y++)
    {
        cout << "      ";
        for (int x = 0; x < 3; x++)
            cout << intTocharColorType(rc[up][y][x]) << " ";
        cout << endl;
    }

    // left, front, right face
    for (int y = 0; y < 3; y++)
    {
        for (int x = 0; x < 9; x++)
        {
            int i = 0;
            switch (x / 3)
            {
            case 0:
                i = left;
                break;
            case 1:
                i = front;
                break;
            case 2:
                i = right;
                break;
            }

            cout << intTocharColorType(rc[i][y][x % 3]) << " ";
        }
        cout << endl;
    }

    // bottom face
    for (int y = 0; y < 3; y++)
    {
        cout << "      ";
        for (int x = 0; x < 3; x++)
            cout << intTocharColorType(rc[bottom][y][x]) << " ";
        cout << endl;
    }

    cout << "Back face:" << endl;
    // back face
    for (int y = 0; y < 3; y++)
    {
        cout << "      ";
        for (int x = 0; x < 3; x++)
            cout << intTocharColorType(rc[back][y][x]) << " ";
        cout << endl;
    }
}

int main()
{
    VideoCapture cam = VideoCapture(1);

    if (!cam.isOpened())
    {
        cout << "No cam detected...";
        return 0;
    }

    Mat frame; bool ok = 1;
    int rc[6][3][3]; int nr = 0, front = 0, up = 1, left, right, bottom, back;
    int SquareCheck = 1, ColorCheck = 1, RubikGame = 1;
    while (ok)
    {
        cam >> frame;
        if (frame.empty()) // Check if the frame is empty
        {
            cout << "Error: Captured frame is empty!" << endl;
            break;
        }
        
        vector<vector<Point>> sq;
        vector<int> colorType;


        if (RubikGame)
        {
            // translate the squares to data as input (while also filtering the extras)
            findSquaresAlt(frame, sq, colorType);
            filterExtraContours(sq, colorType);

            // debug -> mark individually the squares gathered as data
            if (SquareCheck)
                drawSquares(frame, sq);

            // debug -> render individually the color gathered as data (minus white)
            if (ColorCheck)
            {
                Mat color = CombineColorFilterResults(frame, vector<int>({ Blue, Yellow, Red, Green, Orange }));
                addWeighted(frame, 0.5, color, 1.0, 0.0, frame);
            }

            // translate and position the squares into a 3x3 matrix (and repeat it to the console)
            int a[3][3];
            translateFaceColor(frame, sq, colorType, a);
            for (int i = 0; i < 3; i++)
            {
                for (int j = 0; j < 3; j++)
                    cout << intTocharColorType(a[i][j]) << " ";

                cout << endl;
            }

            // user will position the rubik cube and press 'r' when sure to save input
            if (waitKey(500) == 'r')
            {
                for (int i = 0; i < 3; i++)
                    for (int j = 0; j < 3; j++)
                        rc[nr][i][j] = a[i][j];

                cout << "Input saved (" << nr+1 << "/6)" << endl;
                cout << endl;

                nr++;
            }

            switch (nr)
            {
            case 0: // front
                cout << "f: White / u: Orange";
                break;
            case 1: // up
                cout << "f: Orange / u: Yellow";
                break;
            case 2: // bottom
                cout << "f: Red / u: White";
                break;
            case 3: // right
                cout << "f: Green / u: Orange";
                break;
            case 4: // left
                cout << "f: Blue / u: Orange";
                break;
            case 5: //back
                cout << "f: Yellow / u: Orange";
                break;
            case 6:
                cout << "You're done!";
                front = color2side(White);
                up = color2side(Orange);

                ok = 0;
                break;
            }

            cout << endl;
        }
        
        imshow("Webcam", frame);
        
        if (waitKey(300) == 'q') ok = 0; // Stop the program 
    }

    cam.release();
    destroyAllWindows();

    if (RubikGame)
    {
        ReadRubik(front, up, rc);
    }

    int nrc[6][3][3]; copyRC(nrc, rc);
    while (RubikGame)
    {
        char move;
        cout << "Your move... ";
        cin >> move;

        if (move == '0') RubikGame = 0;
        if (move == '/')
        {
            copyRC(rc, nrc);
            cout << "Copied!" << endl;
        }

        actionRubik(front, up, rc, move);
        cout << endl;
        ReadRubik(front, up, rc);
    }
    return 0;
}
